git config --global user.name "اسمك"	تعيين اسم المستخدم لجميع المستودعات.
git config --global user.email "بريدك@مثال.com"	تعيين البريد الإلكتروني لجميع المستودعات.
git config --list	عرض جميع إعدادات Git الحالية.
git init	إنشاء مستودع (Repository) جديد داخل المجلد الحالي.
git clone <URL>	نسخ مستودع موجود من GitHub إلى جهازك.
📁 ثانياً: أوامر الملفات والتعديلات
الأمر	الوظيفة
git status	عرض حالة الملفات في المشروع (معدلة – جديدة – محذوفة).
git add <اسم الملف>	إضافة ملف واحد لمنطقة الانتظار (Staging area).
git add .	إضافة جميع الملفات المعدّلة.
git rm <اسم الملف>	حذف ملف من المستودع.
git mv <الاسم القديم> <الاسم الجديد>	إعادة تسمية أو نقل ملف.
💾 ثالثاً: أوامر الحفظ (Commits)
الأمر	الوظيفة
git commit -m "رسالة التعديل"	حفظ التغييرات مع رسالة توضيحية.
git log	عرض سجل الكوميتات (Commits).
git show <رقم الكوميت>	عرض تفاصيل كوميت محدد.
git diff	عرض الفرق بين الملفات قبل تنفيذ الكوميت.
🌿 رابعاً: أوامر الفروع (Branches)
الأمر	الوظيفة
git branch	عرض جميع الفروع الموجودة.
git branch <اسم>	إنشاء فرع جديد.
git checkout <اسم الفرع>	الانتقال إلى فرع معين.
git checkout -b <اسم الفرع>	إنشاء فرع جديد والانتقال إليه مباشرة.
git merge <اسم الفرع>	دمج فرع محدد مع الفرع الحالي.
git branch -d <اسم الفرع>	حذف فرع محلي.
☁️ خامساً: أوامر الرفع والتنزيل من GitHub (Remote)
الأمر	الوظيفة
git remote -v	عرض المستودعات البعيدة المرتبطة بالمشروع.
git remote add origin <URL>	ربط المستودع المحلي بـ GitHub.
git push origin main	رفع التعديلات إلى الفرع الرئيسي في GitHub.
git pull origin main	جلب التحديثات من GitHub إلى جهازك ودمجها.
git fetch	جلب التحديثات فقط دون دمجها.
🔁 سادساً: أوامر متقدمة ومفيدة
الأمر	الوظيفة
git reset <اسم الملف>	إزالة ملف من الـ Staging قبل الحفظ.
git reset --hard <commit-id>	الرجوع إلى نسخة معينة من المشروع (تحذف التعديلات).
git stash	حفظ التعديلات مؤقتًا دون كوميت.
git stash pop	استرجاع التعديلات المؤقتة.
git log --oneline	عرض سجل الكوميتات بشكل مختصر.
